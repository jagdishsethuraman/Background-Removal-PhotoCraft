import browzwear as bw

def place_artwork():
    garment = bw.get_active_garment()
    piece = garment.get_piece("Front_Body")
    artwork = piece.add_artwork(
        file_path="/Users/ragavij/Documents/Developer/photocraft_vstitcher_studio/output/sample_hoodie_cutout.png",
        x_pos=150.0, y_pos=200.0
    )
    artwork.set_scale(1.0)
    artwork.set_material_type(bw.MaterialType.EMBROIDERY)

place_artwork()
